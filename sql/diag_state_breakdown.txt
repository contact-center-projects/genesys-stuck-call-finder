/* =====================================================================
   diag_state_breakdown.sql - CCPulse says 5 waiting in a queue, our
   query found 2. This shows ALL live calls tied to that queue right
   now, with their real party state - no "no agent leg / in acd queue"
   filter applied - so we can see exactly what the other 3 are doing.
   ===================================================================== */
SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED;
SET NOCOUNT ON;

DECLARE @queue varchar(255) = N'LeumiCall_SECTOR_2';

WITH Parties AS (
    SELECT p.CALLID,
        MAX(CASE WHEN p.AGENTID > 0 THEN p.AGENTID END)        AS AgentID,
        MAX(CASE WHEN p.AGENTID > 0 THEN p.STATE END)          AS AgentState,
        MAX(CASE WHEN p.ENDPOINTTYPE = 4 THEN 1 ELSE 0 END)    AS InAcdQueue,
        MAX(CASE WHEN p.AGENTID IS NULL OR p.AGENTID = 0
                 THEN p.ENDPOINTDN END)                        AS QueueEndpointDN
    FROM G_PARTY p WITH (NOLOCK)
    WHERE p.TERMINATED IS NULL
    GROUP BY p.CALLID
),
Live AS (
    SELECT c.CALLID, c.CALLANI, c.CREATED AS CallCreated,
           pt.AgentID, pt.AgentState, pt.InAcdQueue, pt.QueueEndpointDN
    FROM G_CALL_ACTIVE c
    JOIN Parties pt ON pt.CALLID = c.CALLID
),
Tagged AS (
    -- unbounded on purpose (diagnostic only) - every history row ever
    -- written for this call that matches the queue value.
    SELECT DISTINCT h.CALLID
    FROM G_USERDATA_HISTORY h WITH (NOLOCK)
    JOIN Live l ON l.CALLID = h.CALLID
    WHERE h.VALUE = @queue
)
SELECT
    l.CALLID, l.CALLANI, l.CallCreated, l.AgentID, l.AgentState, l.InAcdQueue, l.QueueEndpointDN,
    CASE
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 1 THEN 'in acd queue'
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 0 THEN 'no agent leg'
        WHEN l.AgentState = 2                          THEN 'ringing agent (2)'
        WHEN l.AgentState = 4                          THEN 'waiting/reserved (4)'
        WHEN l.AgentState = 3                          THEN 'connected (3)'
        ELSE CONCAT('agent state=', l.AgentState)
    END AS StateDesc
FROM Live l
LEFT JOIN Tagged t ON t.CALLID = l.CALLID
WHERE t.CALLID IS NOT NULL OR l.QueueEndpointDN = @queue
ORDER BY StateDesc, l.CallCreated;

-- Same thing, just counted per state, for a quick total-vs-CCPulse check.
;WITH Parties AS (
    SELECT p.CALLID,
        MAX(CASE WHEN p.AGENTID > 0 THEN p.AGENTID END)        AS AgentID,
        MAX(CASE WHEN p.AGENTID > 0 THEN p.STATE END)          AS AgentState,
        MAX(CASE WHEN p.ENDPOINTTYPE = 4 THEN 1 ELSE 0 END)    AS InAcdQueue,
        MAX(CASE WHEN p.AGENTID IS NULL OR p.AGENTID = 0
                 THEN p.ENDPOINTDN END)                        AS QueueEndpointDN
    FROM G_PARTY p WITH (NOLOCK)
    WHERE p.TERMINATED IS NULL
    GROUP BY p.CALLID
),
Live AS (
    SELECT c.CALLID, pt.AgentState, pt.InAcdQueue, pt.QueueEndpointDN
    FROM G_CALL_ACTIVE c
    JOIN Parties pt ON pt.CALLID = c.CALLID
),
Tagged AS (
    SELECT DISTINCT h.CALLID
    FROM G_USERDATA_HISTORY h WITH (NOLOCK)
    JOIN Live l ON l.CALLID = h.CALLID
    WHERE h.VALUE = @queue
)
SELECT
    CASE
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 1 THEN 'in acd queue'
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 0 THEN 'no agent leg'
        WHEN l.AgentState = 2                          THEN 'ringing agent (2)'
        WHEN l.AgentState = 4                          THEN 'waiting/reserved (4)'
        WHEN l.AgentState = 3                          THEN 'connected (3)'
        ELSE CONCAT('agent state=', l.AgentState)
    END AS StateDesc,
    COUNT(*) AS CallCount
FROM Live l
LEFT JOIN Tagged t ON t.CALLID = l.CALLID
WHERE t.CALLID IS NOT NULL OR l.QueueEndpointDN = @queue
GROUP BY
    CASE
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 1 THEN 'in acd queue'
        WHEN l.AgentState IS NULL AND l.InAcdQueue = 0 THEN 'no agent leg'
        WHEN l.AgentState = 2                          THEN 'ringing agent (2)'
        WHEN l.AgentState = 4                          THEN 'waiting/reserved (4)'
        WHEN l.AgentState = 3                          THEN 'connected (3)'
        ELSE CONCAT('agent state=', l.AgentState)
    END
ORDER BY StateDesc;
